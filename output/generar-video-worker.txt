// CLOUDFLARE WORKER — generar-video v2
// Deploy manually via Cloudflare dashboard → Workers → Create Worker → paste this script
//
// Required secrets (Worker Settings → Variables → Secret):
//   OPENROUTER_API_KEY
//   SUPABASE_URL
//   SUPABASE_SERVICE_ROLE_KEY
//
// Two routes:
//   POST /  → Tito builds prompt → submits to OpenRouter /api/v1/videos → returns { job_id }
//   GET  /?job_id=xxx → polls OpenRouter → returns { status, video_url }
//
// After deploy: copy the worker URL and replace the TODO comment in Chat00ImgVid.jsx

const corsHeaders = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Methods': 'POST, GET, OPTIONS',
  'Access-Control-Allow-Headers': 'Content-Type',
}

// Maps briefingState.orientacion values → OpenRouter aspect_ratio
const ASPECT_RATIO_MAP = {
  horizontal: '16:9',
  vertical: '9:16',
  cuadrado: '1:1',
}

// Default resolution per model
const MODEL_RESOLUTION = {
  'x-ai/grok-imagine-video': '720p',
  'bytedance/seedance-2.0': '720p',
}

export default {
  async fetch(request, env) {
    // CORS preflight
    if (request.method === 'OPTIONS') {
      return new Response('ok', { headers: corsHeaders })
    }

    const json = (data, status = 200) =>
      new Response(JSON.stringify(data), {
        headers: { ...corsHeaders, 'Content-Type': 'application/json' },
        status,
      })

    // ── Validate secrets ───────────────────────────────────────────────────
    const { OPENROUTER_API_KEY: apiKey, SUPABASE_URL: supabaseUrl, SUPABASE_SERVICE_ROLE_KEY: serviceKey } = env
    if (!apiKey || !supabaseUrl || !serviceKey) {
      return json({ error: 'Missing required secrets: OPENROUTER_API_KEY, SUPABASE_URL, SUPABASE_SERVICE_ROLE_KEY' }, 500)
    }

    const orHeaders = {
      Authorization: `Bearer ${apiKey}`,
      'Content-Type': 'application/json',
      'HTTP-Referer': 'https://r7signal.com',
      'X-Title': 'R7Signal',
    }

    // ══════════════════════════════════════════════════════════════════════
    // GET /?job_id=xxx  →  Poll video status
    // ══════════════════════════════════════════════════════════════════════
    if (request.method === 'GET') {
      const url = new URL(request.url)
      const jobId = url.searchParams.get('job_id')
      if (!jobId) {
        return json({ error: 'job_id query param is required' }, 400)
      }

      const pollResp = await fetch(`https://openrouter.ai/api/v1/videos/${jobId}`, {
        headers: orHeaders,
      })

      if (!pollResp.ok) {
        const err = await pollResp.text()
        return json({ error: `OpenRouter poll error: ${pollResp.status} — ${err}` }, 502)
      }

      const pollData = await pollResp.json()
      // pollData: { id, status, polling_url, unsigned_urls?, usage?, error? }

      if (pollData.status === 'completed') {
        return json({
          status: 'completed',
          video_url: pollData.unsigned_urls?.[0] ?? null,
        })
      }

      if (pollData.status === 'failed') {
        return json({ status: 'failed', error: pollData.error ?? 'Generation failed' })
      }

      // pending | in_progress
      return json({ status: pollData.status })
    }

    // ══════════════════════════════════════════════════════════════════════
    // POST /  →  Submit video generation job
    // ══════════════════════════════════════════════════════════════════════
    if (request.method === 'POST') {
      let body
      try {
        body = await request.json()
      } catch {
        return json({ error: 'Invalid JSON body' }, 400)
      }

      const {
        user_id,
        menu_numero,
        estilo_id,
        objetos,
        vista,
        orientacion,
        duracion = 5,
        imagen_url,      // URL de imagen generada previamente (PATH A img2video)
        imagen_b64,      // base64 si viene de upload (PATH B)
      } = body

      if (!user_id || !menu_numero) {
        return json({ error: 'user_id and menu_numero are required' }, 400)
      }

      const sbHeaders = {
        Authorization: `Bearer ${serviceKey}`,
        apikey: serviceKey,
      }

      // ── 1. Fetch prompt_base ───────────────────────────────────────────
      let promptBase = null
      if (estilo_id) {
        const res = await fetch(
          `${supabaseUrl}/rest/v1/estilos_imagen?id=eq.${estilo_id}&select=prompt_base`,
          { headers: sbHeaders }
        )
        if (res.ok) {
          const data = await res.json()
          if (data.length > 0) promptBase = data[0].prompt_base
        }
      }

      // ── 2. Fetch Tito slug ────────────────────────────────────────────
      const titoRes = await fetch(
        `${supabaseUrl}/rest/v1/menu_items?menu_numero=eq.${menu_numero}&tipo=eq.tito&modulo_id=eq.592f5894-41d8-4191-90b7-b1e47e166626&select=modelo_id&limit=1`,
        { headers: sbHeaders }
      )
      const titoData = await titoRes.json()
      const titoSlug = titoData[0]?.modelo_id
      if (!titoSlug) return json({ error: 'Tito model not found for this menu' }, 404)

      // ── 3. Fetch Asun·Video slug ──────────────────────────────────────
      const asunRes = await fetch(
        `${supabaseUrl}/rest/v1/menu_items?menu_numero=eq.${menu_numero}&tipo=eq.asun_video&modulo_id=eq.4425984b-704e-472a-9b7c-965ae51c55b8&select=modelo_id&limit=1`,
        { headers: sbHeaders }
      )
      const asunData = await asunRes.json()
      const asunSlug = asunData[0]?.modelo_id
      if (!asunSlug) return json({ error: 'Asun·Video model not found for this menu' }, 404)

      // ── 4. Tito: build prompt_final ───────────────────────────────────
      const titoSystem = `You are a professional video prompt engineer for ${asunSlug}.
Combine the style base and user inputs into a single optimized English prompt.
Include: subject description, motion style, camera movement (pan, tilt, zoom, orbit), 
lighting, atmosphere. Be specific and cinematographic.
Output ONLY the final prompt — no explanations, no preamble, no quotes.`

      const titoUser = `Style base: ${promptBase || 'cinematic photorealistic style'}
Subject / objects: ${objetos || 'a scene'}
Camera view: ${vista || 'medium shot'}
Duration context: ${duracion} seconds`

      const titoResp = await fetch('https://openrouter.ai/api/v1/chat/completions', {
        method: 'POST',
        headers: orHeaders,
        body: JSON.stringify({
          model: titoSlug,
          messages: [
            { role: 'system', content: titoSystem },
            { role: 'user', content: titoUser },
          ],
          max_tokens: 350,
          temperature: 0.7,
        }),
      })

      if (!titoResp.ok) {
        const err = await titoResp.text()
        return json({ error: `Tito error: ${titoResp.status} — ${err}` }, 502)
      }

      const titoJson = await titoResp.json()
      const promptFinal = titoJson.choices?.[0]?.message?.content?.trim()
      if (!promptFinal) return json({ error: 'Tito returned empty prompt' }, 502)

      // ── 5. Build /api/v1/videos payload ───────────────────────────────
      const aspectRatio = ASPECT_RATIO_MAP[orientacion] ?? '16:9'
      const resolution = MODEL_RESOLUTION[asunSlug] ?? '720p'

      const videoPayload = {
        model: asunSlug,
        prompt: promptFinal,
        duration: Number(duracion),
        aspect_ratio: aspectRatio,
        resolution,
      }

      // Image reference: use frame_images for img2video
      const refImageUrl = imagen_url || imagen_b64 || null
      if (refImageUrl) {
        videoPayload.frame_images = [
          {
            type: 'image_url',
            image_url: { url: refImageUrl },
            frame_type: 'first_frame',
          },
        ]
      }

      // ── 6. Submit to OpenRouter /api/v1/videos ────────────────────────
      const submitResp = await fetch('https://openrouter.ai/api/v1/videos', {
        method: 'POST',
        headers: orHeaders,
        body: JSON.stringify(videoPayload),
      })

      if (!submitResp.ok) {
        const err = await submitResp.text()
        return json({ error: `OpenRouter submit error: ${submitResp.status} — ${err}` }, 502)
      }

      const submitData = await submitResp.json()
      // submitData: { id, polling_url, status: "pending" }

      return json({
        job_id: submitData.id,
        status: submitData.status,  // "pending"
        prompt_final: promptFinal,  // útil para debug / mostrar al usuario
      }, 202)
    }

    return json({ error: 'Method not allowed' }, 405)
  },
}
